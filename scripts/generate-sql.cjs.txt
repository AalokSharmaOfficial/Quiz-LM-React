// scripts/generate-sql.cjs
const fs = require('fs');
const path = require('path');

// Define file paths
const jsonFilePath = path.join(__dirname, '..', 'public', 'questions.json');
const sqlFilePath = path.join(__dirname, '..', 'public', 'populate_db.sql');

console.log('Starting SQL generation script...');
console.log(`Reading questions from: ${jsonFilePath}`);

try {
    // Read the JSON file
    const questionsData = fs.readFileSync(jsonFilePath, 'utf-8');
    const questions = JSON.parse(questionsData);

    console.log(`Found ${questions.length} questions.`);

    // Helper function to format values for SQL
    const formatValue = (value) => {
        if (value === null || value === undefined) {
            return 'NULL';
        }
        if (typeof value === 'string') {
            // Escape single quotes by doubling them
            return `'${value.replace(/'/g, "''")}'`;
        }
        if (typeof value === 'number' || typeof value === 'boolean') {
            return value;
        }
        if (Array.isArray(value)) {
            // Format as a PostgreSQL array literal
            const formattedArray = value.map(item => `"${item.replace(/"/g, '""')}"`).join(',');
            return `ARRAY[${formattedArray}]`;
        }
        if (typeof value === 'object') {
            // Format as a JSON string literal for JSONB columns
            return `'${JSON.stringify(value).replace(/'/g, "''")}'`;
        }
        return 'NULL';
    };

    // Generate SQL INSERT statements
    const sqlStatements = questions.map(q => {
        const columns = Object.keys(q).join(', ');
        const values = Object.values(q).map(formatValue).join(', ');
        return `INSERT INTO questions (${columns})\nVALUES\n(${values});`;
    }).join('\n\n');

    const finalSqlScript = `-- This script populates the \`questions\` table with ${questions.length} entries.\n-- Generated based on questions.json\n\n${sqlStatements}\n`;

    // Write to the .sql file
    fs.writeFileSync(sqlFilePath, finalSqlScript, 'utf-8');

    console.log(`Successfully generated ${questions.length} INSERT statements.`);
    console.log(`SQL script saved to: ${sqlFilePath}`);

} catch (error) {
    console.error('An error occurred:', error);
    process.exit(1);
}
